cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
include(vendor/pico-sdk/pico_sdk_init.cmake)
project(uart_usb_engine C CXX ASM)
pico_sdk_init()
add_executable(uart_usb_engine main.c)
target_include_directories(uart_usb_engine PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(uart_usb_engine pico_stdlib hardware_uart hardware_irq tinyusb_device)
target_compile_options(uart_usb_engine PRIVATE -Wall -Wextra)
pico_enable_stdio_usb(uart_usb_engine 0)
pico_enable_stdio_uart(uart_usb_engine 0)
pico_add_extra_outputs(uart_usb_engine)
